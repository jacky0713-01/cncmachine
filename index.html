function updateTradingText() {
let lfList = [];
let haveList = [];
const sortedKeys = Object.keys(itemsData).sort((a, b) => itemsData[a].id - itemsData[b].id);
sortedKeys.forEach(key => {
const item = itemsData[key];
const isUnlocked = item.astral || item.divine || item.void || (item.astralCount > 0);
// LF (未收集)
if (!isUnlocked) {
lfList.push(`Astral/Divine/Void "${item.name}"`);
}
// I have (多餘數量)
if (item.astralCount > 1) {
const extraCount = item.astralCount - 1;
haveList.push(`Astral "${item.name}" x${extraCount}`);
}
});
let tradeStr = `LF\n`;
tradeStr += lfList.length > 0 ? lfList.join('\n') : 'None';
tradeStr += `\n\nI have \n`;
tradeStr += haveList.length > 0 ? haveList.join('\n') : 'None';
tradeStr += `\n\nNo tradeback\nDM`;
